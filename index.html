Install

git clone https://github.com/teslasolar/light.git
cd light
npm link
light repl

CLI

light run program.light          # run a file
light eval "🔻 4.0 42 👁"        # eval inline
light template add a=10 b=32    # parameterized template
light std max a=10 b=20          # stdlib function
light pipe add:a=5:b=3 phi      # chain functions
light compile file.prasm         # PRASM → LIGHT transpile
light mcp sys                    # MCP tool interface
light ops                        # opcode table
light list                       # all functions
light repl                       # interactive REPL

Architecture

mcp(              ← 12 tools
  api(            ← createMCP, createVM
    cli(          ← run/eval/template/std/pipe/compile/mcp/ops/repl
      cmd(        ← dispatch
        sys(      ← 888 registers, 7 rings, parse+exec
          params( ← templates + stdlib + user fns
            light ← 38 emoji opcodes, all primes
          )
        )
      )
    )
  )
)
